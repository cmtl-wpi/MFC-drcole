% Thermocapillary (Marangoni) migration of a droplet -- publication schematic.
% Companion to examples/2D_thermocapillary_migration (Samareh et al. 2014, Test Case 1).
%
% Compile:  pdflatex mechanism_schematic.tex   (or lualatex / tectonic)
%
% Physics: a drop sits in an imposed linear temperature field (cold floor, hot ceiling).
% Surface tension falls with temperature, so sigma is high on the cold (bottom) side and
% low on the hot (top) side -- drawn as a graded interface (dark/heavy = high sigma). The
% tangential sigma-gradient is the Marangoni stress; it drags interfacial fluid from hot
% to cold (top -> bottom) and drives an internal recirculation, so by reaction the drop
% migrates toward the hot ceiling (U). Annotations are split for legibility: the graded
% interface and sigma callouts sit on the left, the resulting Marangoni stress on the
% right; the stress is of course axisymmetric about the migration axis.

\documentclass[border=6pt]{standalone}
\usepackage{amsmath}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, decorations.markings}

% restrained, print-friendly palette
\definecolor{coldc}{RGB}{46,86,149}
\definecolor{hotc}{RGB}{171,57,52}
\definecolor{marac}{RGB}{31,107,99}
\definecolor{flowc}{RGB}{124,131,143}
\definecolor{inkc}{RGB}{28,28,30}

\begin{document}
\begin{tikzpicture}[
    >=Stealth, font=\small, line cap=round, line join=round,
    lead/.style={inkc!55, line width=0.4pt},                 % thin annotation leaders
    marr/.style={marac, line width=1.05pt,                   % Marangoni tangential arrow
      decoration={markings, mark=at position 0.52 with {\arrow{Stealth[length=4.2pt]}}},
      postaction=decorate},
  ]
  \def\R{1.5}                                                % droplet radius

  %--- imposed temperature field: cold floor (bottom) -> hot ceiling (top) ---------------
  \shade[bottom color=coldc!16, top color=hotc!18] (-2.7,-3.5) rectangle (2.7,3.5);
  \draw[inkc!35, line width=0.5pt] (-2.7,-3.5) rectangle (2.7,3.5);
  \node[hotc]  at (0, 3.14) {hot wall};
  \node[coldc] at (0,-3.14) {cold wall};

  %--- imposed-gradient direction (left margin) ------------------------------------------
  \draw[->, inkc!55, line width=0.7pt] (-2.95,-1.7) -- (-2.95,1.7);
  \node[inkc!75, rotate=90] at (-3.14,0) {$\nabla T$};

  %--- droplet: lightly shaded interior + interface graded by local sigma ----------------
  \shade[inner color=white, outer color=inkc!7] (0,0) circle (\R);
  % interface -- each arc segment shaded/weighted by local sigma (high on cold pole):
  \foreach \a in {0,3,...,357}{
    \pgfmathsetmacro{\t}{(sin(\a)+1)/2}       % 0 at cold pole (bottom), 1 at hot pole (top)
    \pgfmathsetmacro{\dk}{92-48*\t}           % darkness -- high sigma reads dark
    \pgfmathsetmacro{\lw}{2.0-1.1*\t}         % weight   -- high sigma reads heavy
    \draw[inkc!\dk, line width=\lw pt] (\a:\R) arc[start angle=\a, end angle=\a+4, radius=\R];
  }

  %--- sigma magnitude callouts (left) ---------------------------------------------------
  \draw[lead] (150:\R) -- (-1.7,1.0);
  \node[hotc,  anchor=east, font=\footnotesize] at (-1.66,1.0) {low $\sigma$};
  \draw[lead] (210:\R) -- (-1.7,-1.0);
  \node[coldc, anchor=east, font=\footnotesize] at (-1.66,-1.0) {high $\sigma$};

  %--- Marangoni tangential stress (right): interfacial fluid driven hot -> cold ---------
  \draw[marr] (62:1.82)  arc[start angle=62,  end angle=8,   radius=1.82];
  \draw[marr] (-8:1.82)  arc[start angle=-8,  end angle=-62, radius=1.82];
  \draw[lead] (1.82,0) -- (2.28,0.02);
  \node[marac, anchor=west, align=left] at (2.24,0) {Marangoni\\stress $\;\nabla_{\!s}\sigma$};

  %--- internal recirculation (subordinate): surface down the sides, return up the core --
  % left loop CCW, right loop CW -- outer flanks descend, core rises (arrowheads on the core)
  \draw[flowc, line width=0.7pt, ->]
    (-0.70,0.906) arc[start angle=100, end angle=400,  x radius=0.44, y radius=0.92];
  \draw[flowc, line width=0.7pt, ->]
    ( 0.70,0.906) arc[start angle=80,  end angle=-220, x radius=0.44, y radius=0.92];

  %--- drop migration toward the hot side (upward) ---------------------------------------
  \draw[inkc, line width=2.1pt, ->] (0,-0.2) -- (0,2.02);
  \node[inkc, anchor=west] at (0.16,1.78) {$U$};

\end{tikzpicture}
\end{document}
